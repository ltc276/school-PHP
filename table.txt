-- 1. Tạo database
CREATE DATABASE csdlot;

-- Chọn database
USE csdlot;


-- =========================================
-- 2. Tạo bảng TAIKHOAN
-- =========================================

CREATE TABLE TAIKHOAN (
    iduser INT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(50) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    hodem VARCHAR(50),
    ten VARCHAR(50),
    phanquyen VARCHAR(20),
    landangnhapcuoi DATETIME
);

-- Tạo chỉ mục cho password
CREATE INDEX idx_password ON TAIKHOAN(password);


-- =========================================
-- 3. Tạo bảng CONGTY
-- =========================================

CREATE TABLE CONGTY (
    idcty INT AUTO_INCREMENT PRIMARY KEY,
    tencty VARCHAR(100) NOT NULL,
    diachi VARCHAR(255),
    dienthoai VARCHAR(20),
    fax VARCHAR(20)
);


-- =========================================
-- 4. Tạo bảng SANPHAM
-- =========================================

CREATE TABLE SANPHAM (
    idsp INT AUTO_INCREMENT PRIMARY KEY,
    tensp VARCHAR(100) NOT NULL,
    gia DECIMAL(15,2),
    mota TEXT,
    hinh VARCHAR(255),
    giamgia DECIMAL(5,2),
    idcty INT,

    CONSTRAINT fk_sanpham_congty
    FOREIGN KEY (idcty) REFERENCES CONGTY(idcty)
);
